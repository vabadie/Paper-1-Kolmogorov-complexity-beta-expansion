$\N_2 := \{2,3,4,\ldots\}$

POLICY: We treat the case $\base \in (1,2)$.

Story justification for the upcoming definition. It is not possible to convert a prefix of the greedy expansion of a number $s$ to the prefix of the greedy expansion but in another base, except if some very specific conditions hold between the bases (established for integer bases in Brattka, to be investigated for nonintegers). Make the example with base $2$ and $3$. Hence given a prefix the greedy expansion of a number $s$ in base $\base_1$, we can only hope to build a function that gives a list of possible prefixes in base $\base_2$, i.e., such a function takes one element as input, and outputs a set of elements. This exactly falls into the framework introduced previously. as a multivalued function in Brattka

For $\base \in (1,\infty)$, we define the base $\base$ evaluation function $ \ev_\base : \subseteq \N^\N \to [0,1]$ as
\begin{equation}
    \ev_\base(\xf) = \sum_{i=1}^\infty \xf_i \base^{-i},
\end{equation}
for all $\xf \in \dom( \ev_\base) := \digs$. We set $\eval :=  \ev_\base|^{[0,1]}$, i.e., $\eval$ is the evaluation function restricted to those elements of $\digs$ that are a $\base$-expansion of a real number in $[0,1]$. For $f : X \to \N^\N$, $A \subseteq X$ and $n \in \N$, we define
\begin{equation}
    f(A)|_n := \{ \xf|_n : \xf \in f(A)\}.
\end{equation}

\begin{definition} Let $\base_1,\base_2 \in (1,\infty)$, and $f : \N \to \N$ be a nondecreasing function. The prefix base conversion multifunction is denoted by $c_{\base_1,\base_2}(\cdot | f)$ the function that, upon presentation of a prefix of length $n \in \N$ of any expansion in base $\base_1$ as input, outputs the possible corresponding prefixes of length $f(n)$ of the greedy expansion in base $\base_2$. In mathematical symbols, this translates to
    \begin{equation}
        \begin{array}{cclc}
            c_{\base_1,\base_2}(\cdot | f) : & \subseteq \N^\ast & \rightrightarrows & \N^\ast\\
            & x & \mapsto &  \left[\grd[\base_2]\circ\eval[\base_1](x\N^\N)\right]_{1:f(|x|)}.
        \end{array}
    \end{equation}
    Note that $\dom (c_{\base_1,\base_2}(\cdot|f))$ is the set of finite words in $\N^\ast$ that are a prefix of some element of $\dom \eval[\base_1]$.

\end{definition}

We now want to establish an upper bound on $\card (c_{\base_1,\base_2}(x|f))$, for $x \in \dom (c_{\base_1,\base_2}(\cdot|f))$.  Note that $\eval[\base_1](x \N^\N)$ is an interval, of length at most $\frac{\base_1^{-|x|}}{\base_1-1}$. We next establish an upper bound on $\# \grd[\base_2](I)_{1:n}$, where $I \subset [0,1]$ is a closed interval and $n \in \N$. The upper bound will notably depend on $|I|$.

\begin{theorem}
    Let $\base \in (1,\infty)$ and $I \subseteq [0,1]$ be a closed interval. Then,
    \begin{equation}\label{eq:bound number canonical sequences interval}
        \# \gf_b(I)_{1:n} \leq |I|\frac{\lceil b \rceil}{b-1} b^{n} + (\lceil b \rceil -b)\sum_{i=1}^n (G_b^i(\inf I) - G_b^i(\sup I)).
    \end{equation}
\end{theorem}
\begin{proof}
    Let $b \in (1,2)$, $I \subseteq [0,1]$ be a closed interval, and $n \in \N$. First note that for $A,B \subseteq [0,1]$ disjoint,
    \begin{equation}\label{eq:greedy expansions union}
        \gf_b(A )_{1:n} \subseteq \gf_b(A \cup B)_{1:n} \subseteq \gf_b(A)_{1:n} \cup \gf_b(B)_{1:n}.
    \end{equation}
    Next, let us define 
    \begin{align}
        s_n &:= \delta_b(\gf_b(\inf I)_{1:n}),\\
        t_n &:= \delta_b(\gf_b(\sup I)_{1:n}),
    \end{align}
    and $I_n := [s_n,t_n]$. We prove that (a) $\gf_b(I_n)_{1:n} = \gf_b(I)_{1:n}$ and (b) that (\ref{eq:bound number canonical sequences interval}) holds for $I_n$.
    \begin{enumerate}[label=(\alph*)]
        \item Note that $I_n \cup (s_n, \sup I] = [s_n, \inf I) \cup I = [s_n, \sup I]$. We show that $\gf_b([s_n, \sup I])_{1:n} =\gf_b(I_n)_{1:n}$ and that $\gf_b([s_n, \sup I])_{1:n} =\gf_b(I)_{1:n}$, yielding $\gf_b(I_n)_{1:n} =\gf_b(I)_{1:n}$. 
        
        First note that, by \ref{eq:greedy expansions union}, we have
        \begin{align}
            \gf_b(I)_{1:n} &\subseteq \gf_b([s_n, \sup I])_{1:n} \subseteq \gf_b(I)_{1:n} \cup \gf_b([s_n,\inf I))_{1:n}.\\
            \gf_b(I_n)_{1:n} &\subseteq \gf_b([s_n, \sup I])_{1:n} \subseteq \gf_b(I_n)_{1:n} \cup \gf_b((t_n,\sup I])_{1:n}.
        \end{align}
        We prove that (i) $\gf_b([s_n,\inf I))_{1:n} \subseteq \gf_b(I)_{1:n}$ and (ii) $\gf_b(I_n)_{1:n} \subseteq \gf_b((t_n,\sup I])_{1:n}$.

        \begin{enumerate}[label=(\roman*)]
            \item Suppose that there exists $x \in \gf_b((s_n, \inf I])_{1:n} \backslash \ \gf_b(I)_{1:n}$. Note that, in particular, $x \neq \gf_b(s_n)$. As $\inf I - \delta_b(x) < \inf I - s_n \leq b^{-n}$, $x$ is the $n$-prefix of a $b$-expansion of $\inf I$. But since $\delta_b(x) > s_n = \delta_b(\gf_b(\inf I)_{1:n})$, then by Lemma \ref{GR-lem:greedy expansion preserves order}, $x >_L \gf_b(\inf I)_{1:n}$, which is a contradiction.
            \item Suppose that there exists $x \in \gf_b((t_n, \sup I])_{1:n} \backslash \ \gf_b(I_n)_{1:n}$. Note that, in particular, $x \neq \gf_b(s_n)$. As $\sup I - \delta_b(x) < \sup I - t_n \leq b^{-n}$, $x$ is the $n$-prefix of a $b$-expansion of $\sup I$. But since $\delta_b(x) > t_n = \delta_b(\gf_b(\sup I)_{1:n})$, then by Lemma \ref{GR-lem:greedy expansion preserves order}, $x >_L \gf_b(\sup I)_{1:n}$, which is a contradiction.
        \end{enumerate}
        \item We find a law on the number of elements of $\gf_b(I_n)_{1:n}$ based on $\gf_b(I_n)_{1:n-1}$.
    \end{enumerate}
\end{proof}

    Let $b_1 \in (1,\infty)$ and $b_2 \in \N_2$. Then,
    \begin{equation}
       \left\lceil\frac{(b_2-1)b_2^{f(|x|)}}{b_1^{{|x|}}}\right\rceil \leq N_{b_1,b_2}(x|f) \leq \left\lceil\frac{(b_2-1)b_2^{f(|x|)}}{b_1^{{|x|}}}\right\rceil +1.
    \end{equation}
\end{theorem}
