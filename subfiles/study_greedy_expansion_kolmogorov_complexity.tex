$\N_2 := \{2,3,4,\ldots\}$

Story justification for the upcoming definition. It is not possible to convert a prefix of the greedy expansion of a number $s$ to the prefix of the greedy expansion but in another base, except if some very specific conditions hold between the bases (established for integer bases in Brattka, to be investigated for nonintegers). Make the example with base $2$ and $3$. Hence given a prefix the greedy expansion of a number $s$ in base $b_1$, we can only hope to build a function that gives a list of possible prefixes in base $b_2$, i.e., such a function takes one element as input, and outputs a set of elements. This exactly falls into the framework introduced previously. as a multivalued function in Brattka. 


\begin{definition} Let $b_1,b_2 \in (1,\infty)$, and $f : \N \to \N$ be a nondecreasing function. The prefix base conversion multifunction is denoted by $c_{b_1,b_2}(\cdot | f)$ the function that, upon presentation of a prefix of length $n \in \N$ of any expansion in base $b_1$ as input, outputs the possible corresponding prefixes of length $f(n)$ of the greedy expansion in base $b_2$. In mathematical symbols, this translate to
    \begin{equation}
        \begin{array}{cclc}
            c_{b_1,b_2}(\cdot | f) : & \subseteq \N^\ast & \rightrightarrows & \N^\ast\\
            & x & \mapsto &  \gf_{b_2}(\delta_{b_1}(x\N^\N))_{1:f(|x|)}.
        \end{array}
    \end{equation}
    We also define $N_{b_1,b_2}(\cdot|f) : \subseteq \N^\ast \to \N$ as
    \begin{equation}
        N_{b_1,b_2}(x|f) := \#c_{b_1,b_2}(x| f), \ \forall x \in \dom(c_{b_1,b_2}(\cdot|f)) := \rg(\gf_{b_1}).
    \end{equation}
\end{definition}

We now want to establish an upper bound on $N_{b_1,b_2}(x|f)$.  Note that $\delta_{b_1}(x \N^\N)$ is an interval, of length $\frac{\beta^{-|x|}}{\beta-1}$. We next establish a general uppoer bound on $\# \gf_{b_2}(I)_{1:n}$, where $I$ is an interval and $n \in \N$. The upper bound will notably depend on $|I|$.

\begin{lemma}
    Let $b \in (1,\infty)$, and $I \subseteq [0,1]$ be an interval. Then,
    \begin{equation}
        \# \gf_b(I)_{1:n} \leq 1 + |I|b^{n} + \sum_{i=1}^n (G_b^i(\inf I) - G_b^i(\sup I)).
    \end{equation}
\end{lemma}
\noindent We denote 
\begin{align}
    s_i &:= \delta_b(\gf_b(\inf I)_{1:i}) \\
    t_i &:= \delta_b(\gf_b(\sup I)_{1:i})
\end{align}
A variant of the right term of the above equation is
\begin{align}
    \sum_{i=1}^n (G_b^i(\inf I) - G_b^i(\sup I)) &= \sum_{i=1}^n \left[\beta^i (\inf I - s_i) - \beta^i(\sup I - t_i) \right]\\
    &= \sum_{i=1}^n \beta^i (t_i - s_i - |I|) 
\end{align}

\begin{theorem}
    Let $b_1 \in (1,\infty)$ and $b_2 \in \N_2$. Then,
    \begin{equation}
       \left\lceil\frac{(b_2-1)b_2^{f(|x|)}}{b_1^{{|x|}}}\right\rceil \leq N_{b_1,b_2}(x|f) \leq \left\lceil\frac{(b_2-1)b_2^{f(|x|)}}{b_1^{{|x|}}}\right\rceil +1.
    \end{equation}
\end{theorem}

\begin{theorem}
    Let $b_1,b_2 \in (1,\infty)$, then 
    \begin{equation}
        \left\lceil\frac{(b_2-1)b_2^{f(|x|)}}{b_1^{{|x|}}}\right\rceil \leq N_{b_1,b_2}(x|f) \leq 2^{|x|}.
     \end{equation}
\end{theorem}
